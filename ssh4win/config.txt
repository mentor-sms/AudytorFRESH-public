Host mentor-*
    # Connection
    Port 22
    Protocol 2
    LogLevel INFO

    PreferredAuthentications publickey
    PubkeyAuthentication yes
    StrictHostKeyChecking accept-new
    HashKnownHosts no
    IdentitiesOnly yes

    # !Auth
    PasswordAuthentication no
    ChallengeResponseAuthentication no
    GSSAPIAuthentication no
    HostbasedAuthentication no

    # Burst sessions of one user
    BatchMode yes
    ServerAliveInterval 60
    ServerAliveCountMax 3
    ConnectTimeout 10

    # Features
    Compression no
    ForwardX11 no
    Tunnel no
    RequestTTY no
    GatewayPorts no
    #SendEnv LANG LC_*

    # TCP
    TCPKeepAlive no
    #LocalForward
    #RemoteForward
    #DynamicForward

    # SFTP?
    ForwardAgent no

    # Paths
    UserKnownHostsFile ~/.ssh/known_hosts

    # Modern cipher preferences (Windows OpenSSH compatible)
    Ciphers aes256-gcm@openssh.com,chacha20-poly1305@openssh.com,aes256-ctr,aes192-ctr,aes128-ctr
    MACs hmac-sha2-256-etm@openssh.com,hmac-sha2-512-etm@openssh.com,hmac-sha2-256,hmac-sha2-512
    KexAlgorithms curve25519-sha256@libssh.org,ecdh-sha2-nistp256,ecdh-sha2-nistp384,ecdh-sha2-nistp521
